cmake_minimum_required(VERSION 3.22)
project(Hendrix VERSION 0.1.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS")
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "Build without automatic installation" FORCE)
if(NOT IPLUG2_DIR)
  message(FATAL_ERROR "Set IPLUG2_DIR to the pinned iPlug2 checkout; see build_plugins.sh")
endif()
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(RUST_TARGET_DIR "${CMAKE_BINARY_DIR}/rust")
set(RUST_LIB "${RUST_TARGET_DIR}/release/libhendrix_audio_to_midi.a")
file(GLOB RUST_SOURCES "${CMAKE_SOURCE_DIR}/*.rs")
add_custom_command(OUTPUT "${RUST_LIB}"
  COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${RUST_TARGET_DIR} ${CARGO_EXECUTABLE} build --locked --manifest-path "${CMAKE_SOURCE_DIR}/Cargo.toml" --release --no-default-features
  DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock VERBATIM)
add_custom_target(hendrix-rust DEPENDS "${RUST_LIB}")
add_library(hendrix-core STATIC IMPORTED GLOBAL)
set_target_properties(hendrix-core PROPERTIES IMPORTED_LOCATION "${RUST_LIB}")
add_dependencies(hendrix-core hendrix-rust)
include(${IPLUG2_DIR}/iPlug2.cmake)
find_package(iPlug2 REQUIRED)
set(STAGE "${CMAKE_BINARY_DIR}/plugin-project")
file(MAKE_DIRECTORY "${STAGE}/resources")
foreach(RC main.rc_mac_dlg main.rc_mac_menu resource.h)
 configure_file("${CMAKE_SOURCE_DIR}/${RC}" "${STAGE}/resources/${RC}" COPYONLY)
endforeach()
foreach(PLIST AU VST3 macOS)
 configure_file("${CMAKE_SOURCE_DIR}/Hendrix-${PLIST}-Info.plist" "${STAGE}/resources/Hendrix-${PLIST}-Info.plist" COPYONLY)
endforeach()
configure_file("${CMAKE_SOURCE_DIR}/Hendrix-macOS-MainMenu.xib" "${STAGE}/resources/Hendrix-macOS-MainMenu.xib" COPYONLY)
file(WRITE "${STAGE}/CMakeLists.txt" "iplug_add_plugin(Hendrix SOURCES \"${CMAKE_SOURCE_DIR}/Hendrix.cpp\" \"${CMAKE_SOURCE_DIR}/Hendrix.h\" \"${CMAKE_SOURCE_DIR}/HendrixNative.mm\" WEB_RESOURCES \"${CMAKE_SOURCE_DIR}/index.html\" \"${CMAKE_SOURCE_DIR}/hendrix-fire.png\" \"${CMAKE_SOURCE_DIR}/moonshroom.png\" UI WEBVIEW FORMATS APP VST3 AU LINK hendrix-core)\n")
add_subdirectory("${STAGE}" "${CMAKE_BINARY_DIR}/plugin")
foreach(FORMAT app vst3 au)
 target_include_directories(Hendrix-${FORMAT} PRIVATE "${CMAKE_SOURCE_DIR}")
 target_link_libraries(Hendrix-${FORMAT} PRIVATE "-framework Security" "-framework CoreFoundation" "-framework CoreMIDI" iconv)
endforeach()
